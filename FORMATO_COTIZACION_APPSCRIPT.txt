function onOpen() {
  var ui = DocumentApp.getUi();
  ui.createMenu('⚙️ Automatización')
    .addItem('Crear nuevo documento para Proyecto', 'clonarYEditarDocumento')
    .addToUi();
}

function clonarYEditarDocumento() {
  var uiDoc = DocumentApp.getUi();
  
  // 1. Pedir el Nombre del Cliente (Empresa / Carpeta)
  var respCliente = uiDoc.prompt('Nueva Cotización', '1. Introduce el nombre del CLIENTE / EMPRESA (ej. GBRX):', uiDoc.ButtonSet.OK_CANCEL);
  if (respCliente.getSelectedButton() != uiDoc.Button.OK) return;
  var nombreCliente = respCliente.getResponseText().trim();

  // 2. Pedir el Número/Nombre del Proyecto
  var respProyecto = uiDoc.prompt('Nueva Cotización', '2. Introduce el NÚMERO DE PROYECTO:', uiDoc.ButtonSet.OK_CANCEL);
  if (respProyecto.getSelectedButton() != uiDoc.Button.OK) return;
  var noProyecto = respProyecto.getResponseText().trim();

  // 3. Pedir el Título y Nombre de la persona (ej. Ing. Francisco Manilla)
  var respAtencion = uiDoc.prompt('Nueva Cotización', '3. Introduce el TITULO Y NOMBRE del contacto (ej. Ing. Francisco Manilla):', uiDoc.ButtonSet.OK_CANCEL);
  if (respAtencion.getSelectedButton() != uiDoc.Button.OK) return;
  var tituloNombre = respAtencion.getResponseText().trim();

  // 4. Obtener la fecha actual automáticamente (Formato: DD.MM.AA)
  var hoy = new Date();
  var dia = String(hoy.getDate()).padStart(2, '0');
  var mes = String(hoy.getMonth() + 1).padStart(2, '0');
  var anio = String(hoy.getFullYear()).slice(-2);
  var fechaAutomatica = dia + '.' + mes + '.' + anio;

  // NUEVO ID CORRECTO DE LA CARPETA PROYECTOS_2025
  var idCarpetaRaiz = "16PF5Tjk9U9HcJEjwIibNuNZ6flH7rgG0"; 
  
  try {
    var carpetaRaiz = DriveApp.getFolderById(idCarpetaRaiz);
    
    // Buscar carpeta del Cliente
    var subCarpetasCliente = carpetaRaiz.getFoldersByName(nombreCliente);
    if (!subCarpetasCliente.hasNext()) {
      uiDoc.alert('Error', 'No se encontró la carpeta del cliente "' + nombreCliente + '" dentro de PROYECTOS_2025.', uiDoc.ButtonSet.OK);
      return;
    }
    var carpetaCliente = subCarpetasCliente.next();
    
    // Buscar carpeta del Número de Proyecto
    var subCarpetasProyecto = carpetaCliente.getFoldersByName(noProyecto);
    if (!subCarpetasProyecto.hasNext()) {
      uiDoc.alert('Error', 'No se encontró la carpeta del proyecto "' + noProyecto + '" dentro de ' + nombreCliente, uiDoc.ButtonSet.OK);
      return;
    }
    var carpetaProyecto = subCarpetasProyecto.next();
    
    // Buscar la carpeta DOC
    var subCarpetasDoc = carpetaProyecto.getFoldersByName("DOC");
    if (!subCarpetasDoc.hasNext()) {
      uiDoc.alert('Error', 'No se encontró la subcarpeta "DOC" dentro de este proyecto.', uiDoc.ButtonSet.OK);
      return;
    }
    var carpetaDestinoDoc = subCarpetasDoc.next();
    
    // Clonar el formato de texto en la carpeta de destino
    var archivoFormato = DriveApp.getFileById(DocumentApp.getActiveDocument().getId());
    var copiaArchivo = archivoFormato.makeCopy(noProyecto, carpetaDestinoDoc);
    
    // --- EDICIÓN DEL DOCUMENTO CLONADO ---
    var docCopia = DocumentApp.openById(copiaArchivo.getId());
    
    // Función interna para aplicar los reemplazos en cualquier sección
    var procesarSeccion = function(seccion) {
      if (!seccion) return;
      
      // Reemplazo en texto directo
      seccion.replaceText('{{TITULO_NOMBRE}}', tituloNombre);
      seccion.replaceText('{{EMPRESA}}', nombreCliente);
      seccion.replaceText('{{NO_PROYECTO}}', noProyecto);
      seccion.replaceText('{{FECHA}}', fechaAutomatica);
      
      // Reemplazo dentro de tablas de esa sección
      var tablas = seccion.getTables();
      for (var t = 0; t < tablas.length; t++) {
        var tabla = tablas[t];
        for (var r = 0; r < tabla.getNumRows(); r++) {
          var fila = tabla.getRow(r);
          for (var c = 0; c < fila.getNumCells(); c++) {
            var celda = fila.getCell(c);
            celda.replaceText('{{TITULO_NOMBRE}}', tituloNombre);
            celda.replaceText('{{EMPRESA}}', nombreCliente);
            celda.replaceText('{{NO_PROYECTO}}', noProyecto);
            celda.replaceText('{{FECHA}}', fechaAutomatica);
          }
        }
      }
    };
    
    // 1. Procesar el cuerpo del documento
    procesarSeccion(docCopia.getBody());
    
    // 2. Procesar el encabezado normal
    procesarSeccion(docCopia.getHeader());
    
    // 3. Buscar en secciones especiales (como el encabezado de la primera página diferente)
    var seccionesCuerpo = docCopia.getBody().getParent();
    for (var i = 0; i < seccionesCuerpo.getNumChildren(); i++) {
      var elemento = seccionesCuerpo.getChild(i);
      if (elemento.getType() === DocumentApp.ElementType.HEADER_SECTION) {
        procesarSeccion(elemento.asHeaderSection());
      }
    }
    
    docCopia.saveAndClose();
    
    uiDoc.alert('¡Éxito!', 'Documento creado y personalizado con éxito.\nGuardado en: ' + nombreCliente + ' > ' + noProyecto + ' > DOC.\n\nEnlace: ' + copiaArchivo.getUrl(), uiDoc.ButtonSet.OK);
    
  } catch(e) {
    uiDoc.alert('Error en el proceso', e.toString(), uiDoc.ButtonSet.OK);
  }
}
